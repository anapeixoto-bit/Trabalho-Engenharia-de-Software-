--ACADEMIA DDL

--TABELA ALUNO
CREATE TABLE ALUNO (
ID_ALUNO INT NOT NULL UNIQUE AUTOINCREMENT PRIMARY KEY,
NOME VARCHAR(100) NOT NULL,
COD_ENDERECO INT,
TELEFONE VARCHAR(20) NOT NULL,
DATA_NASCIMENTO DATE NOT NULL,
ALTURA NUMERIC(3, 2),
PESO NUMERIC(3, 2),

CONSTRAINT FK_ALUNO_ENDERECO
	FOREIGN KEY (COD_ENDERECO)
	REFERENCES ENDERECO (ID_ENDERECO)
);

-- TABELA FILHA DE ALUNO
CREATE TABLE ENDERECO (
ID_ENDERECO INT AUTOINCREMENT PRIMARY KEY,
RUA VARCHAR(100) NOT NULL,
NUMERO INT NOT NULL,
CIDADE VARCHAR(30),
ESTADO VARCHAR(10) NOT NULL
);

--TEBELA FILHA DE ATIVIDADE
CREATE TABLE ALUNO_MONITOR (
ID_ALUNO_MONITOR INT NOT NULL UNIQUE AUTOINCREMENT PRIMARY KEY,
NOME VARCHAR(50),
TELEFONE VARCHAR(20)
);

--TABELA INSTRUTOR
CREATE TABLE INSTRUTOR (
ID_INSTRUTOR INT NOT NULL UNIQUE AUTOINCREMENT PRIMARY KEY,
RG VARCHAR(20) NOT NULL UNIQUE,
NOME VARCHAR(100) NOT NULL,
TELEFONE VARCHAR(20) NOT NULL,
DATA_NASCIMENTO DATE,
TITULACAO VARCHAR(100) NOT NULL
);

CREATE TABLE ATIVIDADE (
COD_ATIVIDADE INT AUTOINCREMENT PRIMARY KEY,
TIPO_ATIVIDADE VARCHAR(50) NOT NULL UNIQUE,
ID_ALUNO_MONITOR INT,

CONSTRAINT FK_ATIVIDADE_MONITOR
	FOREIGN KEY (ID_ALUO_MONNITOR)
	REFERENCES ALUNO_MONITOR (ID_ALUNO_MONITOR)
	
);

--TABELA TURMA
CREATE TABLE TURMA (
ID_TURMA INT AUTOINCREMENT PRIMARY KEY,
NUMERO_ALUNOS INT NOT NULL DEFAULT 0,
DURACAO_AULA TIME,
DATA_INICIO DATE,
DATA_TERMINO DATE,
ID_INSTRUTOR INT NOT NULL,
COD_ATIVIDADE INT NOT NULL,
ID_ALUNO_MONITOR INT,

CONSTRAINT FK_TURMA_INSTRUTOR
	FOREIGN KEY (ID_INSTRUTOR)
	REFERENCES INSTRUTOR (ID_INSTRUTOR),
	
CONSTRAINT FK_TURMA_ATIVIDADE
	FOREIGN KEY (COD_ATIVIDADE)
	REFERENCES ATIVIDADE (COD_ATIVIDADE)

CONSTRAINT FK_TURMA_MONITOR
	FOREIGN KEY (ID_ALUNO_MONITOR)
	REFERENCES ALUNO_MONITOR (ID_ALUNO_MONITOR),

);

-- TABELA MATRICULA
CREATE TABLE MATRICULA (
ID_MATRICULA INT NOT NULL UNIQUE PRIMARY KEY,
ID_ALUNO INT NOT NULL,
ID_TURMA INT NOT NULL,

CONSTRAINT FK_MATRICULA_ALUNO
	FOREIGN KEY (ID_ALUNO)
	REFERENCES ALUNO (ID_ALUNO)

CONSTRAINT FK_MATRICULA_TURMA
	FOREIGN KEY (ID_TURMA)
	REFERENCES TURMA (ID_TURMA)
	
CONSTRAINT UK_ALUNO_TURMA
        UNIQUE (ID_ALUNO, ID_TURMA)
);

--TABELA FILHA DE MATRICULA
CREATE TABLE AUSENCIA (
DATA_AUSENCIA DATE NOT NULL UNIQUE PRIMARY KEY,
ID_ALUNO INT,
ID_TURMA INT,

CONSTRAINT FK_ID_ALUNO
	FOREIGN KEY (ID_ALUNO)
	REFERENCES ALUNO (ID_ALUNO)
	
CONSTRAINT FK_ID_TRUMA
	FOREIGN KEY (ID_TURMA)
	REFERENCES TURMA (ID_TURMA)
	
CONSTRAINT UK_AUSENCIA
       UNIQUE (ID_ALUNO, ID_TURMA, DATA_AUSENCIA)
	   
);
